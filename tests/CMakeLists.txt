file(GLOB FDR_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/test_*.cpp)
foreach(test_src IN LISTS FDR_TEST_SOURCES)

  add_executable(${test_name} ${test_src} main.cpp)
  target_link_libraries(${test_name} PRIVATE fdr_core)
  add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()
